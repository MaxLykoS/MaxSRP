// Execute through execute_code in the MaxSRP Editor. Tests the production resolve.
var shader = UnityEngine.Shader.Find("Hidden/MaxSRP/TemporalAA");
if (shader == null || !shader.isSupported) throw new System.Exception("Missing TAA shader");
const int n = 32;
var owned = new System.Collections.Generic.List<UnityEngine.Object>();
var material = new UnityEngine.Material(shader); owned.Add(material);
System.Func<UnityEngine.Texture2D> texture = () => {
    var t = new UnityEngine.Texture2D(n, n, UnityEngine.TextureFormat.RGBAFloat, false, true);
    t.filterMode = UnityEngine.FilterMode.Point; t.wrapMode = UnityEngine.TextureWrapMode.Clamp;
    owned.Add(t); return t;
};
var source = texture(); var history = texture(); var depth = texture(); var previousDepth = texture(); var motion = texture(); var capability = texture();
var color = new UnityEngine.RenderTexture(n,n,0,UnityEngine.RenderTextureFormat.ARGBHalf,UnityEngine.RenderTextureReadWrite.Linear);
var meta = new UnityEngine.RenderTexture(n,n,0,UnityEngine.RenderTextureFormat.RGFloat,UnityEngine.RenderTextureReadWrite.Linear);
owned.Add(color); owned.Add(meta); color.Create(); meta.Create();
var read = texture();
var gpu = UnityEngine.GL.GetGPUProjectionMatrix(UnityEngine.Matrix4x4.Perspective(60,1,.1f,100),true);
var clip = gpu * new UnityEngine.Vector4(0,0,-5,1);
float device = clip.z / clip.w;
// This Editor fixture targets D3D11 device depth. Ordinary-Z shader branches are compiled separately.
var reports = new System.Collections.Generic.List<string>();
var savedRT = UnityEngine.RenderTexture.active;
try {
    for (int test=0;test<12;++test) {
        var src = new UnityEngine.Color[n*n]; var hist = new UnityEngine.Color[n*n];
        var dep = new UnityEngine.Color[n*n]; var old = new UnityEngine.Color[n*n]; var vel = new UnityEngine.Color[n*n];
        var caps = new UnityEngine.Color[n*n];
        for(int y=0;y<n;++y) for(int x=0;x<n;++x) {
            int i=y*n+x;
            float value = test==1 ? (((x+y)&1)==0 ? .2f : .3f) : test==2 ? (x<n/2 ? .2f : .8f) : .25f;
            if(test==8) value=.2f+x*.01f;
            if(test==9) value=.2f+y*.01f;
            if(test>=10) value=(((x+y)&1)==0 ? .2f : .3f);
            src[i]=new UnityEngine.Color(value,value,value,.37f);
            float h=test==2 ? value : .25f;
            hist[i]=new UnityEngine.Color(h,h,h,.37f);
            if(test==8 || test==9) hist[i]=new UnityEngine.Color(value+.01f,value+.01f,value+.01f,.37f);
            dep[i]=new UnityEngine.Color(device,0,0,0);
            old[i]=new UnityEngine.Color(test==3 ? 3 : 5,16,0,0);
            vel[i]=new UnityEngine.Color(test==4 ? 2 : 0,0,5,test==5 ? -1 : 1);
            if(test==8) vel[i]=new UnityEngine.Color(1f/n,0,5,1);
            if(test==9) vel[i]=new UnityEngine.Color(0,1f/n,5,1);
            if(test>=10) vel[i]=UnityEngine.Color.clear;
            caps[i]=new UnityEngine.Color(.5f,.5f,1,test==11 ? 0 : 1);
            if(test==3 || test==4 || test==5 || test==6) hist[i]=new UnityEngine.Color(.8f,.8f,.8f,1);
            if(test==7) { dep[i]=UnityEngine.Color.clear; old[i]=new UnityEngine.Color(0,16,0,0); vel[i]=new UnityEngine.Color(0,0,0,-1); }
        }
        source.SetPixels(src); source.Apply(); history.SetPixels(hist); history.Apply();
        depth.SetPixels(dep); depth.Apply(); previousDepth.SetPixels(old); previousDepth.Apply(); motion.SetPixels(vel); motion.Apply();
        capability.SetPixels(caps);capability.Apply();
        history.filterMode=UnityEngine.FilterMode.Bilinear;
        var cmd=new UnityEngine.Rendering.CommandBuffer();
        cmd.SetGlobalTexture("_TaaSource",source); cmd.SetGlobalTexture("_TaaHistory",history);
        cmd.SetGlobalTexture("_TaaDepth",depth); cmd.SetGlobalTexture("_TaaPreviousDepth",previousDepth); cmd.SetGlobalTexture("_TaaMotion",motion);
        cmd.SetGlobalTexture("_GBuffer1",capability);
        cmd.SetGlobalVector("_TaaSize",new UnityEngine.Vector4(n,n,1f/n,1f/n));
        cmd.SetGlobalVector("_TaaSourceRect",new UnityEngine.Vector4(1,1,0,0));
        cmd.SetGlobalVector("_TaaSettings",new UnityEngine.Vector4(test==6 ? 0 : 1,.94f,1.25f,0));
        cmd.SetGlobalVector("_MaxTaaJitter",UnityEngine.Vector4.zero);
        cmd.SetGlobalVector("_WorldSpaceCameraPos",UnityEngine.Vector4.zero);
        cmd.SetGlobalVector("unity_OrthoParams",UnityEngine.Vector4.zero);
        cmd.SetGlobalMatrix("unity_MatrixInvVP",gpu.inverse);
        cmd.SetGlobalMatrix("unity_MatrixV",UnityEngine.Matrix4x4.identity);
        cmd.SetGlobalMatrix("_MaxPreviousView",UnityEngine.Matrix4x4.identity);
        cmd.SetGlobalMatrix("_MaxPreviousVP",gpu); cmd.SetGlobalMatrix("_MaxUnjitteredVP",gpu);
        cmd.SetRenderTarget(new UnityEngine.Rendering.RenderTargetIdentifier[]{color,meta},new UnityEngine.Rendering.RenderTargetIdentifier(UnityEngine.Rendering.BuiltinRenderTextureType.None));
        cmd.SetViewport(new UnityEngine.Rect(0,0,n,n));
        UnityEditor.ShaderUtil.SetAsyncCompilation(cmd,false);
        cmd.DrawProcedural(UnityEngine.Matrix4x4.identity,material,0,UnityEngine.MeshTopology.Triangles,3);
        UnityEditor.ShaderUtil.RestoreAsyncCompilation(cmd);
        UnityEngine.Graphics.ExecuteCommandBuffer(cmd); cmd.Release();
        UnityEngine.RenderTexture.active=color; read.ReadPixels(new UnityEngine.Rect(0,0,n,n),0,0); read.Apply();
        var pixels=read.GetPixels(); double error=0,noise=0; float maxError=0;
        for(int i=0;i<pixels.Length;++i) {
            var p=pixels[i];
            if(float.IsNaN(p.r)||float.IsInfinity(p.r)) throw new System.Exception("Nonfinite TAA output");
            float e=UnityEngine.Mathf.Abs(p.r-src[i].r); maxError=UnityEngine.Mathf.Max(maxError,e);
            error+=(p.r-src[i].r)*(p.r-src[i].r); noise+=(p.r-.25f)*(p.r-.25f);
            if(UnityEngine.Mathf.Abs(p.a-.37f)>.001f) throw new System.Exception("Source alpha lost");
        }
        if(test!=1 && test!=10 && maxError>.002f) throw new System.Exception("TAA fixture "+test+" error "+maxError);
        if((test==1 || test==10) && noise/pixels.Length>=.001) throw new System.Exception("Noise did not converge");
        UnityEngine.RenderTexture.active=meta; read.ReadPixels(new UnityEngine.Rect(0,0,n,n),0,0); read.Apply();
        float count=read.GetPixel(n/2,n/2).g;
        if((test==3||test==4||test==5||test==6||test==11)&&UnityEngine.Mathf.Abs(count-1)>.01f) throw new System.Exception("Rejected history retained count");
        reports.Add("case="+test+" maxError="+maxError+" rmsNoise="+System.Math.Sqrt(noise/pixels.Length)+" count="+count);
    }
} finally {
    UnityEngine.RenderTexture.active=savedRT;
    foreach(var o in owned) UnityEngine.Object.DestroyImmediate(o);
    UnityEditor.SceneView.RepaintAll();
}
var folder="D:/UnityProject/MaxSRP/OfflineBaseline/TemporalAA";
System.IO.Directory.CreateDirectory(folder);
System.IO.File.WriteAllLines(folder+"/gpu_fixtures.txt",reports.ToArray());
return reports.ToArray();
