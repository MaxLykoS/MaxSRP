UnityEditor.EditorApplication.CallbackFunction timeout=null;
capture=(cmd,camera,source,resolved,motion)=>{
    if(camera.cameraType!=UnityEngine.CameraType.SceneView)return;
    MaxSRP.MaxTemporalAAPass.EditorResolveCapture-=capture;UnityEditor.EditorApplication.update-=timeout;
    var targets=new System.Collections.Generic.List<UnityEngine.RenderTexture>();
    var materials=new System.Collections.Generic.List<UnityEngine.Material>();
    var labels=new[]{"oldRaw32","correctedRaw32","oldMean64","correctedMean64","correctedFlatMean64","oldFlatMean64","correctedMean32","oldMean32"};
    int previousBlocker=UnityEngine.Shader.GetGlobalInteger("_PcssBlockerSamples"),previousFilter=UnityEngine.Shader.GetGlobalInteger("_PcssFilterSamples");
    UnityEditor.ShaderUtil.SetAsyncCompilation(cmd,false);
    for(int v=0;v<labels.Length;++v){
        var rt=new UnityEngine.RenderTexture(700,356,0,UnityEngine.RenderTextureFormat.ARGBFloat,UnityEngine.RenderTextureReadWrite.Linear);
        rt.Create();targets.Add(rt);
        var mat=new UnityEngine.Material(shader);materials.Add(mat);

            var block=new UnityEngine.MaterialPropertyBlock();
            block.SetFloat("_MainLightShadowsAvailable",1);block.SetMatrix("_ShadowReceiverView",camera.worldToCameraMatrix);
            cmd.SetGlobalInteger("_PcssBlockerSamples",v<2 || v>=6 ? 16 : 64);cmd.SetGlobalInteger("_PcssFilterSamples",v<2 || v>=6 ? 32 : 64);
            block.SetVector("_PcssProbe",new UnityEngine.Vector4(1f/count,v==4 || v==5 ? 1 : 0,v==1 || v==3 || v==4 || v==6 ? 1 : 0,0));

            cmd.DrawProcedural(UnityEngine.Matrix4x4.identity,mat,count>1 ? 1 : 0,UnityEngine.MeshTopology.Triangles,3,1,block);
        }
    }
    cmd.SetGlobalInteger("_PcssBlockerSamples",previousBlocker);cmd.SetGlobalInteger("_PcssFilterSamples",previousFilter);

            }finally{UnityEngine.Object.DestroyImmediate(tex);}
