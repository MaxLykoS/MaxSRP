    float dx=.5f*(jittered.x/jittered.w-unjittered.x/unjittered.w);

    h.Valid=true;h.Sample=5;h.PreviousProjection=camera.projectionMatrix;h.LastRenderTime=UnityEngine.Time.realtimeSinceStartupAsDouble;h.Rotation=camera.transform.rotation;
